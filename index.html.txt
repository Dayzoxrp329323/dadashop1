import path from "path";
import { fileURLToPath } from "url";

import jwt from "jsonwebtoken";
import Database from "better-sqlite3";

const JWT_SECRET = process.env.JWT_SECRET || "dev-secret-change-me";

fs.mkdirSync(path.join(__dirname, "public/uploads"), { recursive: true });

const db = new Database(path.join(__dirname, "marketplace.db"));
db.pragma("journal_mode = WAL");
db.exec(`
CREATE TABLE IF NOT EXISTS users(
 id INTEGER PRIMARY KEY AUTOINCREMENT,
 name TEXT NOT NULL,
 email TEXT UNIQUE NOT NULL,
 password TEXT NOT NULL,
 wallet_cents INTEGER NOT NULL DEFAULT 0,
 created_at TEXT DEFAULT CURRENT_TIMESTAMP
);
CREATE TABLE IF NOT EXISTS products(
 id INTEGER PRIMARY KEY AUTOINCREMENT,
 seller_id INTEGER NOT NULL,

CREATE TABLE IF NOT EXISTS orders(
 id INTEGER PRIMARY KEY AUTOINCREMENT,
 buyer_id INTEGER NOT NULL,
 seller_id INTEGER NOT NULL,
 product_id INTEGER NOT NULL,
 amount_cents INTEGER NOT NULL,
 status TEXT NOT NULL DEFAULT 'paid',
 created_at TEXT DEFAULT CURRENT_TIMESTAMP
);
`);

const upload = multer({ dest: path.join(__dirname, "public/uploads/") });
app.use(express.json());

function auth(req,res,next){
  const h=req.headers.authorization||"";
  const token=h.startsWith("Bearer ")?h.slice(7):null;
  if(!token) return res.status(401).json({error:"Connexion requise"});
  try { req.user=jwt.verify(token,JWT_SECRET); next(); }
  catch { res.status(401).json({error:"Session invalide"}); }
}

app.post("/api/register", async (req,res)=>{
  const {name,email,password}=req.body;
  if(!name||!email||!password||password.length<6) return res.status(400).json({error:"Nom, email et mot de passe (6 caractères minimum) requis"});
  try{
    const hash=await bcrypt.hash(password,12);
    const info=db.prepare("INSERT INTO users(name,email,password) VALUES(?,?,?)").run(name,email.toLowerCase(),hash);
    const user={id:Number(info.lastInsertRowid),name,email:email.toLowerCase()};
    const token=jwt.sign(user,JWT_SECRET,{expiresIn:"7d"});
    res.json({token,user});
  }catch(e){res.status(400).json({error:"Cet email est déjà utilisé"});}
});

app.post("/api/login", async (req,res)=>{
  const {email,password}=req.body;
  const u=db.prepare("SELECT * FROM users WHERE email=?").get((email||"").toLowerCase());
  if(!u || !(await bcrypt.compare(password||"",u.password))) return res.status(401).json({error:"Email ou mot de passe incorrect"});
  const user={id:u.id,name:u.name,email:u.email};
  res.json({token:jwt.sign(user,JWT_SECRET,{expiresIn:"7d"}),user});
});

app.get("/api/me",auth,(req,res)=>{
  const u=db.prepare("SELECT id,name,email,wallet_cents FROM users WHERE id=?").get(req.user.id);
  res.json(u);
});

app.get("/api/products",(req,res)=>{
  const q=(req.query.q||"").trim();
  const rows=q ? db.prepare(`SELECT p.*,u.name seller FROM products p JOIN users u ON u.id=p.seller_id WHERE p.sold=0 AND (p.title LIKE ? OR p.description LIKE ?) ORDER BY p.id DESC`).all(`%${q}%`,`%${q}%`)
               : db.prepare(`SELECT p.*,u.name seller FROM products p JOIN users u ON u.id=p.seller_id WHERE p.sold=0 ORDER BY p.id DESC`).all();
  res.json(rows);
});

app.post("/api/products",auth,upload.single("image"),(req,res)=>{
  const {title,description,price,category}=req.body;
  const cents=Math.round(Number(price)*100);
  if(!title||!description||!category||!Number.isFinite(cents)||cents<=0) return res.status(400).json({error:"Informations d'annonce invalides"});
  const image=req.file?"/uploads/"+req.file.filename:"";
  const info=db.prepare("INSERT INTO products(seller_id,title,description,price_cents,category,image) VALUES(?,?,?,?,?,?)")
    .run(req.user.id,title,description,cents,category,image);
  res.json({id:Number(info.lastInsertRowid)});
});

/*
  DEMO CHECKOUT:
  This endpoint creates an order using the logged-in buyer's wallet balance.
  It does NOT charge a real card. For production, replace this part with a
  verified PayPal/Stripe checkout and webhook, then credit the seller only
  after the provider confirms payment.
*/
app.post("/api/buy/:id",auth,(req,res)=>{
  const product=db.prepare("SELECT * FROM products WHERE id=? AND sold=0").get(req.params.id);
  if(!product) return res.status(404).json({error:"Article indisponible"});
  if(product.seller_id===req.user.id) return res.status(400).json({error:"Tu ne peux pas acheter ta propre annonce"});
  const buyer=db.prepare("SELECT * FROM users WHERE id=?").get(req.user.id);
  if(buyer.wallet_cents<product.price_cents) return res.status(402).json({error:"Solde portefeuille insuffisant (mode démo)"});
  const tx=db.transaction(()=>{
    db.prepare("UPDATE users SET wallet_cents=wallet_cents-? WHERE id=?").run(product.price_cents,buyer.id);
    db.prepare("UPDATE users SET wallet_cents=wallet_cents+? WHERE id=?").run(product.price_cents,product.seller_id);
    db.prepare("UPDATE products SET sold=1 WHERE id=?").run(product.id);
    return db.prepare("INSERT INTO orders(buyer_id,seller_id,product_id,amount_cents) VALUES(?,?,?,?)").run(buyer.id,product.seller_id,product.id,product.price_cents);
  });
  const info=tx();
  res.json({ok:true,orderId:Number(info.lastInsertRowid)});
});

app.get("/api/orders",auth,(req,res)=>{
  const rows=db.prepare(`SELECT o.*,p.title,p.image,u.name seller FROM orders o JOIN products p ON p.id=o.product_id JOIN users u ON u.id=o.seller_id WHERE o.buyer_id=? ORDER BY o.id DESC`).all(req.user.id);
  res.json(rows);
});

app.post("/api/wallet/demo-credit",auth,(req,res)=>{
  const cents=Math.round(Number(req.body.amount)*100);
  if(!Number.isFinite(cents)||cents<=0||cents>10000000) return res.status(400).json({error:"Montant invalide"});
  db.prepare("UPDATE users SET wallet_cents=wallet_cents+? WHERE id=?").run(cents,req.user.id);
  res.json({ok:true});
});

app.get("*",(req,res)=>res.sendFile(path.join(__dirname,"public/index.html")));
app.listen(PORT,()=>console.log(`Marketplace: http://localhost:${PORT}`));
